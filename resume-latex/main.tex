\section{Projects}

\project
{Sparsh-Vaani: Medicine Label Reader for Differently-Abled Patients}
{Raspberry Pi, ESP32, OCR}
{Built an assistive system that reads medicine labels using a
camera and OCR, stores details in Excel, and controls a
six-compartment pill dispenser through ESP32, with voice alerts,
reminders, and IoT notifications.}

\project
{Smart PillCare: IoT Medicine Dispenser for Elderly}
{ESP32, Blynk IoT}
{Developed an automatic dispenser with pre-alert reminders,
pill dispensing, remaining-tablet count, Blynk app notifications,
and email alerts to caregivers for missed doses; documented in
an IEEE-format paper.}

\project
{AquaVision: Smart Water Quality Monitoring}
{ESP32, Blynk Cloud}
{Built an IoT system to monitor TDS, pH, turbidity, and
temperature, measure water level using HC-SR04, and send
mobile notifications and buzzer alerts.}

\project
{Disaster Management Robot}
{Arduino UNO, ESP32}
{Designed a rescue robot with fire and gas detection, obstacle
avoidance, Bluetooth control, IoT monitoring, and an automatic
medicine dispenser; won first prize in an Embedded Systems
Hackathon.}

\project
{Bluetooth-Controlled RC Car}
{Arduino UNO, HC-05, L298N}
{Built a four-motor car controlled from a smartphone app
(forward, backward, left, right, and stop) using an HC-05 module
and L298N motor driver.}

\project

{ESP32, IR Sensors}
